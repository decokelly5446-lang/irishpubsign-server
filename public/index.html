// ── PHOTO STRIP CONVEYOR BELT ── (function() { var track = document.querySelector('.strip-track'); if (!track) return; // Clone all items and append for seamless loop var origItems = Array.from(track.querySelectorAll('.strip-item')); origItems.forEach(function(item) { track.appendChild(item.cloneNode(true)); }); var speed = 0.5; // pixels per frame — slow conveyor var pos = 0; var paused = false; var halfWidth = null; function getHalfWidth() { // Width of original set (half the total track content) var items = track.querySelectorAll('.strip-item'); var count = origItems.length; var itemW = items[0].offsetWidth + 12; return count * itemW; } function step() { if (!paused) { if (!halfWidth) halfWidth = getHalfWidth(); pos += speed; // When we've scrolled one full set, snap back silently if (pos >= halfWidth) pos -= halfWidth; track.style.transform = 'translateX(-' + pos.toFixed(2) + 'px)'; } requestAnimationFrame(step); } track.addEventListener('mouseenter', function() { paused = true; }); track.addEventListener('mouseleave', function() { paused = false; }); requestAnimationFrame(step); })();
